cmake_minimum_required(VERSION 3.28)
project(foreverdubbed_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "")
include(FetchContent)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sentencepiece URL https://github.com/google/sentencepiece/archive/refs/tags/v0.2.1.tar.gz)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz)
FetchContent_Declare(dr_libs GIT_REPOSITORY https://github.com/mackron/dr_libs.git GIT_TAG dfe8377631000664666519fdb83da193fd8037f4)
# dr_libs is only used by upstream's audio utilities; the app streams saved states.
set(ORT_VERSION 1.23.2)
if(WIN32)
  set(ORT_ARCHIVE onnxruntime-win-x64-${ORT_VERSION}.zip)
elseif(APPLE)
  set(ORT_ARCHIVE onnxruntime-osx-universal2-${ORT_VERSION}.tgz)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  set(ORT_ARCHIVE onnxruntime-linux-aarch64-${ORT_VERSION}.tgz)
else()
  set(ORT_ARCHIVE onnxruntime-linux-x64-${ORT_VERSION}.tgz)
endif()
FetchContent_Declare(ort URL https://github.com/microsoft/onnxruntime/releases/download/v${ORT_VERSION}/${ORT_ARCHIVE})
FetchContent_MakeAvailable(sentencepiece json dr_libs ort)
# Only dependencies are prepared here. Go/cgo compiles PocketTTS.cpp itself.
if(WIN32)
  set(FDB_OS windows)
elseif(APPLE)
  set(FDB_OS darwin)
else()
  set(FDB_OS linux)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
  set(FDB_ARCH arm64)
else()
  set(FDB_ARCH amd64)
endif()
set(FDB_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../.runtime/sdk/${FDB_OS}_${FDB_ARCH}" CACHE PATH "cgo dependency directory")
install(TARGETS sentencepiece-static ARCHIVE DESTINATION "${FDB_SDK}/lib" COMPONENT Native)
install(DIRECTORY ${ort_SOURCE_DIR}/include/ DESTINATION "${FDB_SDK}/include" COMPONENT Native)
install(FILES ${sentencepiece_SOURCE_DIR}/src/sentencepiece_processor.h ${dr_libs_SOURCE_DIR}/dr_wav.h ${dr_libs_SOURCE_DIR}/dr_mp3.h ${dr_libs_SOURCE_DIR}/dr_flac.h DESTINATION "${FDB_SDK}/include" COMPONENT Native)
install(DIRECTORY ${json_SOURCE_DIR}/include/nlohmann DESTINATION "${FDB_SDK}/include" COMPONENT Native)
install(DIRECTORY ${ort_SOURCE_DIR}/lib/ DESTINATION "${FDB_SDK}/lib" COMPONENT Native FILES_MATCHING PATTERN "*.lib" PATTERN "*.so*" PATTERN "*.dylib")
install(DIRECTORY ${ort_SOURCE_DIR}/lib/ DESTINATION . COMPONENT Native FILES_MATCHING PATTERN "*.dll" PATTERN "*.so*" PATTERN "*.dylib")
install(FILES vendor/LICENSE DESTINATION licenses RENAME PocketTTS.cpp.txt COMPONENT Native)
install(FILES ${ort_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME ONNX-Runtime.txt COMPONENT Native)

install(FILES ${sentencepiece_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME SentencePiece.txt COMPONENT Native)
install(FILES ${json_SOURCE_DIR}/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json.txt COMPONENT Native)

install(FILES ${dr_libs_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME dr_libs.txt COMPONENT Native)
